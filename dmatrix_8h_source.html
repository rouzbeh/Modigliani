260 modigliani::DMatrix modigliani::DMatrix::operator~() {
261 int i, j, k, n;
262 int pivrow, tarrow;
263 double pivelt, tarelt;
264 n = row; // Zahl d.Gleich.
265 modigliani::DMatrix aug(n, 2 * col);
266 modigliani::DMatrix inverse(n, col);
267 for (i = 1; i <= n; i++)
268 for (j = 1; j <= col; j++)
269 aug(i, j) = (*this) (i, j);
270 for (i = 1; i <= n; i++)
271 for (j = n + 1; j <= aug.col; j++)
272 if (i == j - n)
273 aug(i, j) = 1; // Einheitsmatrix
274 for (pivrow = 1; pivrow <= n; pivrow++) {
275 pivelt = aug(pivrow, pivrow);
276 if (pivelt == 0) {
277 k = pivrow + 1; // Pivotsuche
278 while (pivelt == 0 && k <= n) {
279 pivelt = aug(k, pivrow);
280 k++;
281 }
282 if (pivelt == 0) {
283 std::cerr
284 << "dmatrix : ERROR - Inversion attempted, but matrix is singular!";
285 exit(1);
286 }
287 } else {
288 k--;
289 modigliani::DMatrix dum(2 * col);
290 for (i = 1; i <= 2 * col; i++)
291 dum(i) = aug(pivrow, i);
292 for (i = 1; i <= 2 * col; i++)
293 aug(pivrow, i) = aug(k, i);
294 for (i = 1; i <= 2 * col; i++)
295 aug(k, i) = dum(i);
296 }
297 }
298
299 for (j = 1; j <= 2 * col; j++)
300 aug(pivrow, j) = aug(pivrow, j) / pivelt;
301 for (tarrow = 1; tarrow <= n; tarrow++)
302 if (tarrow != pivrow) {
303 tarelt = aug(tarrow, pivrow);
304 for (j = 1; j <= 2 * col; j++)
305 aug(tarrow, j) = aug(tarrow, j) - aug(pivrow, j) * tarelt;
306 }
307 for (i = 1; i <= n; i++)
308 for (j = n + 1; j <= aug.col; j++)
309 inverse(i, j - n) = aug(i, j);
310 return (inverse);
311 }
